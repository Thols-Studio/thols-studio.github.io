Export Modal Popup Demo

Interactive test environment for exporting SVG diagrams.

Quick Integration Code:

import { ExportModal } from './export-modal.js';

const modal = ExportModal.init({
  getSvg: () => document.getElementById('sampleSvg'),
  getSourceCode: () => 'flowchart LR\n A[User Action] --> B[Export Modal] --> C(PNG / SVG)',
  getFilename: () => 'my-diagram'
});

document.getElementById('openBtn').addEventListener('click', () => modal.open());
